# Run everything below in one go (self-elevates via UAC when needed). Left as a comment here
# since Setup.ps1 reads this same block and runs it for you - copy/paste this one line alone:
# iex (irm "https://teknoplop.github.io/Setup.ps1")

winget install --id Google.Chrome -e --silent
# Fallback if winget isn't available: Invoke-WebRequest https://dl.google.com/chrome/install/latest/chrome_installer.exe -OutFile chrome_installer.exe; .\chrome_installer.exe
Set-ExecutionPolicy -ExecutionPolicy RemoteSigned -Scope CurrentUser
Invoke-RestMethod -Uri https://get.scoop.sh -OutFile "$env:TEMP\scoop-install.ps1"
& "$env:TEMP\scoop-install.ps1" -RunAsAdmin
scoop install -g git
scoop bucket add extras
scoop bucket add versions
scoop bucket add Sysinternals
scoop install -g main/python 7zip aria2 bat claude claude-code clink cpu-z cryptomator crystaldiskmark furmark gh gpu-z hwinfo msiafterburner notepadplusplus process-explorer ripgrep sudo ventoy vlc vscode wget
scoop config aria2-enabled true
clink autorun install -a
iex (irm "https://teknoplop.github.io/Create-ScoopAppShortcuts.ps1")
iex (irm "https://teknoplop.github.io/Update-Bios.ps1")
# Skips itself (no DDG page opened) if the installed BIOS is under 3 months old, so this
# stays safe to run every time Setup.ps1 runs - e.g. just to refresh desktop shortcuts.

# Why not just use winget for everything, since it's used for Chrome above?
# - Every app in the scoop list also exists in winget, so it's possible.
# - But winget installs one app per command too, so it wouldn't shorten this script.
# - winget is only used pre-Scoop because it's the one thing preinstalled on a clean Windows box.
# - Scoop's only per-line limit is "scoop bucket add" (one bucket per line); app installs
# already batch onto a single "scoop install" line, so there's nothing to gain by switching.
